private void Vinculador_OnInfo(string mensaje)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string>(Vinculador_OnInfo), mensaje);
        return;
    }

    MsnVincular.Text = mensaje;
    toolStripStatusLbl.Text = mensaje;
}

private void Vinculador_OnError(string mensaje)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string>(Vinculador_OnError), mensaje);
        return;
    }

    MsnVincular.Text = mensaje;
    toolStripStatusLbl.Text = mensaje;
    AlertaManager.MostrarAlerta(mensaje, Color.Firebrick, 3, 5);
    txtCodBarras.Clear();
    txtCodBarras.Focus();
}

private void Vinculador_OnInsertadoOk(Dictionary<string, object> data)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<Dictionary<string, object>>(Vinculador_OnInsertadoOk), data);
        return;
    }

    LlenarPrimeraFila(dgvTagList, data); // adapta este nombre si tu grid es otro
    txtCodBarras.Clear();
    txtRFID.Clear();
    _ultimoCodigoBarra = "";
    _ultimoRfid = "";
    toolStripStatusLbl.Text = "Registrado OK";
    AlertaManager.MostrarAlerta("Registrado OK", Color.SeaGreen, 1, 5);
    txtCodBarras.Focus();
}
4. Recepción separada de barcode y RFID
Como tu dispositivo es Zebra y no TS800, la recepción cambia, pero la lógica no.
Cuando llegue barcode:
C#
private void SetCodigoBarras(string codigo)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string>(SetCodigoBarras), codigo);
        return;
    }

    _ultimoCodigoBarra = (codigo ?? "").Trim();
    txtCodBarras.Text = _ultimoCodigoBarra;
    IntentarVincular();
}
Cuando llegue RFID:
C#
private void SetRfid(string rfid)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string>(SetRfid), rfid);
        return;
    }

    _ultimoRfid = (rfid ?? "").Trim().Replace(" ", "");
    txtRFID.Text = _ultimoRfid;
    IntentarVincular();
}
Método que decide si ya puede vincular:
C#
private void IntentarVincular()
{
    if (string.IsNullOrWhiteSpace(_ultimoCodigoBarra)) return;
    if (string.IsNullOrWhiteSpace(_ultimoRfid)) return;

    var request = new VinculacionRequest
    {
        CodigoBarras = _ultimoCodigoBarra,
        Rfid = _ultimoRfid,
        HojaMarcacion = _hojaMarcacionActual,
        CodTrabajador = mCodTrabajador,
        Empresa = mEmpresa
    };

    _vinculador.Enqueue(request);
}
5. Cómo conectarlo con tus eventos Zebra
En tu evento del Zebra, donde hoy clasificas si la lectura parece barcode o RFID, cambia a esto:
C#
private void OnBarcodeEventLector(short eventType, ref string scanData)
{
    try
    {
        string valor = ExtraerDatoLeido(scanData);

        if (string.IsNullOrWhiteSpace(valor))
            return;

        if (PareceRfid(valor))
            SetRfid(valor);
        else
            SetCodigoBarras(valor);
    }
    catch (Exception ex)
    {
        BeginInvoke(new Action(() =>
        {
            toolStripStatusLbl.Text = "Error lectura: " + ex.Message;
        }));
    }
}
