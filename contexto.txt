1) AlertaManager.cs en C#
C#
using System;
using System.Collections.Generic;
using System.Drawing;
using System.Windows.Forms;

namespace DS9908R_App
{
    public static class AlertaManager
    {
        private static readonly List<FormAlerta> alertasActivas = new List<FormAlerta>();
        private static readonly object alertasLock = new object();

        public static void MostrarAlerta(string mensaje, Color color, int tipo = 3, int tiempo = 5)
        {
            var nuevaAlerta = new FormAlerta(mensaje, color, tipo, tiempo);

            if (Application.OpenForms.Count > 0)
            {
                Form form = Application.OpenForms[0];
                if (form.InvokeRequired)
                {
                    form.Invoke(new Action(() => MostrarAlertaInterno(nuevaAlerta)));
                }
                else
                {
                    MostrarAlertaInterno(nuevaAlerta);
                }
            }
            else
            {
                MostrarAlertaInterno(nuevaAlerta);
            }
        }

        private static void MostrarAlertaInterno(FormAlerta alerta)
        {
            lock (alertasLock)
            {
                alerta.StartPosition = FormStartPosition.Manual;
                alerta.Location = new Point(
                    Screen.PrimaryScreen.WorkingArea.Width - alerta.Width - 50,
                    ObtenerPosicion()
                );
                alertasActivas.Add(alerta);
            }

            alerta.Show();
        }

        public static int ObtenerPosicion()
        {
            int pantallaAlto = Screen.PrimaryScreen.WorkingArea.Height;
            int alertaAlto = 70;
            int margen = 10;

            return pantallaAlto - ((alertaAlto + margen) * (alertasActivas.Count + 1));
        }

        public static void RemoverAlerta(FormAlerta alerta)
        {
            lock (alertasLock)
            {
                alertasActivas.Remove(alerta);
                ReorganizarAlertas();
            }
        }

        private static void ReorganizarAlertas()
        {
            lock (alertasLock)
            {
                int alertaAlto = 70;
                int margen = 10;
                int pantallaAlto = Screen.PrimaryScreen.WorkingArea.Height;

                for (int i = 0; i < alertasActivas.Count; i++)
                {
                    alertasActivas[i].Location = new Point(
                        alertasActivas[i].Location.X,
                        pantallaAlto - ((alertaAlto + margen) * (i + 1))
                    );
                }
            }
        }
    }
}

using System;
using System.Drawing;

namespace DS9908R_App

    public class FormAlertaError : Form

        private Label lblDetalle;

            franjaAbajo.Height = 10;

            this.Controls.Add(iconError);
            this.Controls.Add(lblTitulo);
            this.Controls.Add(lblDetalle);
            this.Controls.Add(btnAceptar);
            this.Controls.Add(franjaAbajo);

            this.Load += FormAlertaError_Load;
            this.FormClosed += FormAlertaError_FormClosed;

            ((System.ComponentModel.ISupportInitialize)(this.iconError)).EndInit();
            this.ResumeLayout(false);
        }

        private void FormAlertaError_Load(object sender, EventArgs e)
        {
            if (alertaAbierta)
            {
                this.Close();
                return;
            }

            alertaAbierta = true;

            this.Location = new Point(
                (Screen.PrimaryScreen.WorkingArea.Width - this.Width) / 2,
                (Screen.PrimaryScreen.WorkingArea.Height - this.Height) / 2
            );
        }

        private void BtnAceptar_Click(object sender, EventArgs e)
        {
            alertaAbierta = false;
            this.Close();

            _callback?.Invoke();
        }

        private void FormAlertaError_FormClosed(object sender, FormClosedEventArgs e)
        {
            alertaAbierta = false;
        }
    }
}

4) FormAlerta.cs toast/no bloqueante
C#
using System;
using System.Drawing;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace DS9908R_App
{
    public class FormAlerta : Form
    {
        private int _tiempo = 30;
        private int _desplazamiento = 10;
        private int _destinoY;

        private Label lblMensaje;
        private Panel franjaTop;
        private PictureBox iconError;
        private PictureBox iconOk;
        private PictureBox iconInfo;

        public FormAlerta(string mensaje, Color color, int tipo, int tiempo = 30)
        {
            _tiempo = tiempo;
            InitializeComponent();

            lblMensaje.Text = mensaje;
            lblMensaje.ForeColor = color;
            franjaTop.BackColor = color;
            this.BackColor = ColorTranslator.FromHtml("#fff3cd");
            this.TopMost = true;

            iconError.Visible = false;
            iconOk.Visible = false;
            iconInfo.Visible = false;

            switch (tipo)
            {
                case 1:
                    iconOk.Visible = true;
                    iconOk.Image = SystemIcons.Information.ToBitmap();
                    break;
                case 2:
                    iconInfo.Visible = true;
                    iconInfo.Image = SystemIcons.Question.ToBitmap();
                    break;
                default:
                    iconError.Visible = true;
                    iconError.Image = SystemIcons.Error.ToBitmap();
                    break;
            }

            this.Load += FormAlerta_Load;
        }

        private void InitializeComponent()
        {
            lblMensaje = new Label();
            franjaTop = new Panel();
            iconError = new PictureBox();
            iconOk = new PictureBox();
            iconInfo = new PictureBox();

            ((System.ComponentModel.ISupportInitialize)(iconError)).BeginInit();
            ((System.ComponentModel.ISupportInitialize)(iconOk)).BeginInit();
            ((System.ComponentModel.ISupportInitialize)(iconInfo)).BeginInit();

            this.SuspendLayout();

            this.FormBorderStyle = FormBorderStyle.None;
            this.ShowInTaskbar = false;
            this.ClientSize = new Size(380, 80);
            this.Opacity = 0.97;

            franjaTop.Dock = DockStyle.Top;
            franjaTop.Height = 8;

            iconError.Location = new Point(15, 24);
            iconError.Size = new Size(32, 32);
            iconError.SizeMode = PictureBoxSizeMode.StretchImage;

            iconOk.Location = new Point(15, 24);
            iconOk.Size = new Size(32, 32);
            iconOk.SizeMode = PictureBoxSizeMode.StretchImage;

            iconInfo.Location = new Point(15, 24);
            iconInfo.Size = new Size(32, 32);
            iconInfo.SizeMode = PictureBoxSizeMode.StretchImage;

            lblMensaje.Location = new Point(60, 18);
            lblMensaje.Size = new Size(300, 42);
            lblMensaje.Font = new Font("Segoe UI", 10F, FontStyle.Bold);

            this.Controls.Add(franjaTop);
            this.Controls.Add(iconError);
            this.Controls.Add(iconOk);
            this.Controls.Add(iconInfo);
            this.Controls.Add(lblMensaje);

            ((System.ComponentModel.ISupportInitialize)(iconError)).EndInit();
            ((System.ComponentModel.ISupportInitialize)(iconOk)).EndInit();
            ((System.ComponentModel.ISupportInitialize)(iconInfo)).EndInit();

            this.ResumeLayout(false);
        }

        private async void FormAlerta_Load(object sender, EventArgs e)
        {
            this.Left = Screen.PrimaryScreen.WorkingArea.Width - this.Width - 50;
            this.Top = Screen.PrimaryScreen.WorkingArea.Height;

            _destinoY = AlertaManager.ObtenerPosicion();

            while (this.Top > _destinoY)
            {
                this.Top -= _desplazamiento;
                await Task.Delay(10);
            }

            await Task.Delay(_tiempo * 1000);

            for (double i = 1.0; i >= 0; i -= 0.1)
            {
                this.Opacity = i;
                await Task.Delay(50);
            }

            this.Close();
            AlertaManager.RemoverAlerta(this);
        }
    }
}

C#
private void AlertaErrorMsn(string mensaje, Color color)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string, Color>(AlertaErrorMsn), mensaje, color);
        return;
    }

    toolStripStatusLbl.Text = mensaje;

    using (var alertaError = new FormAlertaError("Error", mensaje, color))
    {
        alertaError.ShowDialog();
    }
}

private void MostrarAlertaErrorBloqueante(string mensaje, Action callback = null)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string, Action>(MostrarAlertaErrorBloqueante), mensaje, callback);
        return;
    }

    toolStripStatusLbl.Text = mensaje;

    if (FormAlertaError.alertaAbierta)
        return;

    using (var alertaError = new FormAlertaError("Error", mensaje, Color.Firebrick, callback))
    {
        alertaError.ShowDialog();
    }
}

private void MostrarAlertaToast(string mensaje, Color color, int tipo = 3, int tiempo = 5)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string, Color, int, int>(MostrarAlertaToast), mensaje, color, tipo, tiempo);
        return;
    }

    toolStripStatusLbl.Text = mensaje;
    AlertaManager.MostrarAlerta(mensaje, color, tipo, tiempo);
}

private void MostrarAlertaOkModal(string titulo, Color color, int tiempo, string descripcion)
{
    if (InvokeRequired)
    {
        BeginInvoke(new Action<string, Color, int, string>(MostrarAlertaOkModal), titulo, color, tiempo, descripcion);
        return;
    }

    using (var alerta = new FormAlertaOk(titulo, color, tiempo, descripcion))
    {
        alerta.ShowDialog();
    }
}
