private void ActualizarTotalCount()
{
    int total = DataGridView1.AllowUserToAddRows
        ? DataGridView1.Rows.Count - 1
        : DataGridView1.Rows.Count;

    if (total < 0)
        total = 0;

    lblTotalCount.Text = total.ToString();

    if (total >= MAX_CACHE_SIZE)
    {
        LimpiarTodo();
    }
}
