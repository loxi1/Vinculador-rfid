private void GenerarConsolidadoDinamico()
{
    try
    {
        if (DataGridView1.Rows.Count == 0)
            return;

        var agrupado = new Dictionary<string, int>();

        foreach (DataGridViewRow row in DataGridView1.Rows)
        {
            if (row.IsNewRow) continue;

            string op = row.Cells["op"].Value?.ToString() ?? "";
            string corte = row.Cells["corte"].Value?.ToString() ?? "";
            string subcorte = row.Cells["subcorte"].Value?.ToString() ?? "";
            string talla = row.Cells["talla"].Value?.ToString() ?? "";

            string key = $"{op}|{corte}|{subcorte}|{talla}";

            if (agrupado.ContainsKey(key))
                agrupado[key]++;
            else
                agrupado[key] = 1;
        }

        DataGridViewConsolidado.Rows.Clear();

        foreach (var item in agrupado)
        {
            var partes = item.Key.Split('|');

            int rowIndex = DataGridViewConsolidado.Rows.Add();

            DataGridViewConsolidado.Rows[rowIndex].Cells["op"].Value = partes[0];
            DataGridViewConsolidado.Rows[rowIndex].Cells["corte"].Value = partes[1];
            DataGridViewConsolidado.Rows[rowIndex].Cells["subcorte"].Value = partes[2];
            DataGridViewConsolidado.Rows[rowIndex].Cells["talla"].Value = partes[3];
            DataGridViewConsolidado.Rows[rowIndex].Cells["cantidad"].Value = item.Value;
        }
    }
    catch (Exception ex)
    {
        SetEstado("Error en consolidado: " + ex.Message, Color.Firebrick);
    }
}
