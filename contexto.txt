public class BDPrenda
{
    public (int codigo, string mensaje) GuardarPrenda(
        string etqt,
        string empresa,
        string usr,
        string rfid,
        string nHoja)
    {
        try
        {
            using (var conn = SybaseHelper.GetConnection())
            {
                conn.Open();

                using (var cmd = conn.CreateCommand())
                {
                    cmd.CommandText = "USP_SAL_EMB_CON_RFID";
                    cmd.CommandType = CommandType.StoredProcedure;

                    cmd.Parameters.Add(new AseParameter("@etqt", etqt));
                    cmd.Parameters.Add(new AseParameter("@empresa", empresa));
                    cmd.Parameters.Add(new AseParameter("@usr", usr));
                    cmd.Parameters.Add(new AseParameter("@rfid", rfid ?? ""));
                    cmd.Parameters.Add(new AseParameter("@nHoja", nHoja ?? ""));

                    using (var reader = cmd.ExecuteReader())
                    {
                        if (reader.Read())
                        {
                            int codigo = Convert.ToInt32(reader["codigo"]);
                            string mensaje = reader["mensaje"].ToString();

                            return (codigo, mensaje);
                        }
                    }
                }
            }

            return (1, "Sin respuesta del SP");
        }
        catch (Exception ex)
        {
            return (99, "Error BD: " + ex.Message);
        }
    }
}

    }

    public void Enqueue(LecturaItem item)

        _cola.Enqueue(item);

    {

    }

    public int Count => _cola.Count;

⚙️ 4. WORKER BACKGROUND (EL CORAZÓN)
C#
public class WorkerProcesador
{
    private readonly ColaLecturas _cola;
    private readonly BDPrenda _bd;

    public event Action<string> OnError;
    public event Action<LecturaItem> OnOk;

    private bool _activo = true;

    public WorkerProcesador(ColaLecturas cola)
    {
        _cola = cola;
        _bd = new BDPrenda();
    }

    public void Iniciar()
    {
        Task.Run(async () =>
        {
            while (_activo)
            {
                if (_cola.TryDequeue(out var item))
                {
                    ProcesarItem(item);
                }
                else
                {
                    await Task.Delay(100); // no CPU al 100%
                }
            }
        });
    }

    private void ProcesarItem(LecturaItem item)
    {
        var resultado = _bd.GuardarPrenda(
            item.CodigoBarra,
            item.Empresa,
            item.Usuario,
            item.RFID,
            item.NHoja);

        if (resultado.codigo == 0)
        {
            OnOk?.Invoke(item);
        }
        else
        {
            item.Intentos++;

            if (item.Intentos < 3)
            {
                _cola.Enqueue(item); // retry
            }
            else
            {
                GuardarOffline(item);
                OnError?.Invoke(resultado.mensaje);
            }
        }
    }

private void GuardarOffline(LecturaItem item)
{
    string path = "offline_queue.json";

    var lista = new List<LecturaItem>();

    if (File.Exists(path))
    {
        var json = File.ReadAllText(path);
        lista = JsonSerializer.Deserialize<List<LecturaItem>>(json);
    }

    lista.Add(item);

    File.WriteAllText(path, JsonSerializer.Serialize(lista, new JsonSerializerOptions
    {
        WriteIndented = true
    }));
}

public void ReprocesarOffline()
{
    string path = "offline_queue.json";

    if (!File.Exists(path)) return;

    var json = File.ReadAllText(path);
    var lista = JsonSerializer.Deserialize<List<LecturaItem>>(json);

    foreach (var item in lista)
    {
        _cola.Enqueue(item);
    }

    File.Delete(path);
}
