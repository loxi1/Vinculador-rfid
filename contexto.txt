1) CodBarras_KeyDown final
Esto sirve tanto para lector que manda Enter como para digitación manual.
C#
private void CodBarras_KeyDown(object sender, KeyEventArgs e)
{
    if (e.KeyCode != Keys.Enter)
        return;

    e.SuppressKeyPress = true;
    e.Handled = true;

    string codigo = (CodBarras.Text ?? "").Trim();

    if (string.IsNullOrWhiteSpace(codigo) || codigo == "Codigo de Barras...")
    {
        toolStripStatusLbl.Text = "Código de barras vacío";
        return;
    }

    _ultimoCodigoBarra = codigo;
    toolStripStatusLbl.Text = "Código de barras leído: " + _ultimoCodigoBarra;

    IntentarVincular();
}

3) SetRfid final
C#
private void SetRfid(string rfid)
{
    BeginInvoke(new Action(() =>
    {
        rfid = (rfid ?? "").Trim().ToUpperInvariant().Replace(" ", "");

        if (string.IsNullOrWhiteSpace(rfid))
            return;

        if (_rfidLeidos.Contains(rfid))
        {
            toolStripStatusLbl.Text = "RFID repetido: " + rfid;
            return;
        }

        _rfidLeidos.Add(rfid);
        _ultimoRfid = rfid;

        if (dgvTagList.Columns.Count == 0)
            ConfigurarGridRfid();

        dgvTagList.Rows.Insert(0, rfid);

        toolStripStatusLbl.Text = "RFID leído: " + rfid;

        IntentarVincular();
    }));
}
4) IntentarVincular() preparado para clientes sin RFID
Aquí está la parte importante.
C#
private void IntentarVincular()
{
    string codigo = (_ultimoCodigoBarra ?? "").Trim();
    string rfid = (_ultimoRfid ?? "").Trim();
    string hojaMarcacion = (nroHM.Text ?? "").Trim();

    if (string.IsNullOrWhiteSpace(codigo))
        return;

    // Modo con RFID: exige ambos
    if (_usarRfid)
    {
        if (string.IsNullOrWhiteSpace(rfid))
            return;
    }
    else
    {
        // Modo solo barras: manda vacío o null
        rfid = "";
    }

    var request = new VinculacionRequest
    {
        CodigoBarras = codigo,
        Rfid = rfid,
        HojaMarcacion = hojaMarcacion,
        CodTrabajador = mCodTrabajador,
        Empresa = mEmpresa,
        UsarRfid = _usarRfid
    };

    _vinculador.Enqueue(request);
}

7) Si SaveRFID(...) en Sybase exige RFID siempre
Aquí hay dos escenarios:
Escenario A: el SP acepta RFID vacío
Entonces no haces nada más.
Escenario B: el SP exige RFID sí o sí
Entonces en modo solo barras no llames SaveRFID(...), sino otro método, por ejemplo:
SaveSoloCodigoBarra(...)
o insertar directo en MySQL
o consultar primero y registrar sin RFID
