El error está clarísimo: en DataGridView1_CellEndEdit todavía estás usando una versión intermedia de la lógica.
Tu código actual intenta usar esto:
C#
_vinculador.BDPrenda
pero VinculadorService no tiene una propiedad pública BDPrenda.
Y además intentas usar:
C#
_bdPrenda
pero ese campo no existe en frmLector.
Por eso salen exactamente estos errores:
CS1061: VinculadorService no contiene definición para BDPrenda
CS0103: _bdPrenda no existe en el contexto
La solución correcta
No uses ni _vinculador.BDPrenda ni _bdPrenda dentro del form.
Debes usar el método público wrapper que agregaste o que debes agregar en VinculadorService.cs.

2) En frmLector.cs corrige DataGridView1_CellEndEdit
Reemplaza toda tu función por esta:
C#
private void DataGridView1_CellEndEdit(object sender, DataGridViewCellEventArgs e)
{
    try
    {
        if (e.RowIndex < 0 || e.ColumnIndex < 0)
            return;

        string nombreColumna = DataGridView1.Columns[e.ColumnIndex].Name;

        if (nombreColumna != "hoja_marcacion")
            return;

        DataGridViewRow fila = DataGridView1.Rows[e.RowIndex];

        string nuevoValor = Convert.ToString(fila.Cells["hoja_marcacion"].Value)?.Trim() ?? "";
        string idRfid = Convert.ToString(fila.Cells["id_rfid"].Value)?.Trim() ?? "";

        if (string.IsNullOrWhiteSpace(idRfid))
        {
            SetEstado("No se puede actualizar: id_rfid vacío.", Color.Firebrick);
            return;
        }

        var where = new Dictionary<string, object>
        {
            { "rfid", idRfid }
        };

        if (!string.IsNullOrWhiteSpace(mCodTrabajador))
            where["fotocheck"] = mCodTrabajador;

        var update = new Dictionary<string, object>
        {
            { "nhoja", nuevoValor }
        };

        if (_vinculador == null)
        {
            SetEstado("El servicio de vinculación no está inicializado.", Color.Firebrick);
            return;
        }

        int resultado = _vinculador.UpdateTimbrado(where, update);

        if (resultado > 0)
        {
            fila.Cells["fecha"].Value = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
            SetEstado("Hoja de marcación actualizada correctamente.", Color.SeaGreen);
        }
        else
        {
            SetEstado("No se actualizó el timbrado.", Color.DarkOrange);
        }
    }
    catch (Exception ex)
    {
        SetEstado("Error al editar: " + ex.Message, Color.Firebrick);
    }
}
