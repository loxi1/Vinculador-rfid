public Tuple<int, string, DataTable> ListarTimbradas(Dictionary<string, object> whereParameters)
{
    int li_return = 0;
    string s_mensaje = string.Empty;
    DataTable tabla = new DataTable();

    if (whereParameters == null)
        whereParameters = new Dictionary<string, object>();

    // Igual que en VB: forzar estado si no viene
    if (!whereParameters.ContainsKey("estado"))
        whereParameters.Add("estado", "EMB"); // cambia "EMB" por tu constante real si ya la tienes en C#

    try
    {
        using (AseConnection conn = _sybase.Connect())
        {
            if (conn == null || conn.State != ConnectionState.Open)
                throw new InvalidOperationException("La conexión a la base de datos no está abierta.");

            string whereClause = string.Join(" AND ",
                whereParameters.Keys.Select(k => $"{k} = @{k}"));

            string sql =
                "SELECT linea, op, nhoja AS hoja_marcacion, corte, sub_corte AS subcorte, " +
                "color, talla, cod_talla, id_talla, fecha, rfid AS id_rfid " +
                "FROM tmp_etiq_timbradas " +
                $"WHERE {whereClause} " +
                "ORDER BY fecha DESC";

            using (AseCommand cmd = new AseCommand(sql, conn))
            {
                foreach (var param in whereParameters)
                    cmd.Parameters.AddWithValue("@" + param.Key, param.Value ?? DBNull.Value);

                using (AseDataReader reader = cmd.ExecuteReader())
                {
                    tabla.Load(reader);
                }
            }

            li_return = tabla.Rows.Count;
            s_mensaje = li_return > 0
                ? "Datos obtenidos correctamente"
                : "No se encontraron resultados";
        }
    }
    catch (Exception ex)
    {
        ss_error = ex.Message;
        LogError("Error en ListarTimbradas", ex);
        li_return = -1;
        s_mensaje = "Error en la consulta";
    }
    finally
    {
        _sybase.Disconnect();
    }

    return Tuple.Create(li_return, s_mensaje, tabla);
}

public int UpdateTimbrado(
    Dictionary<string, object> whereParameters,
    Dictionary<string, object> updateParameters)
{
    int rowsAffected = 0;

    if (whereParameters == null)
        whereParameters = new Dictionary<string, object>();

    if (updateParameters == null || updateParameters.Count == 0)
        return 0;

    if (!whereParameters.ContainsKey("estado"))
        whereParameters.Add("estado", "EMB"); // cambia por tu constante real si corresponde

    try
    {
        using (AseConnection conn = _sybase.Connect())
        {
            if (conn == null || conn.State != ConnectionState.Open)
                throw new InvalidOperationException("La conexión a la base de datos no está abierta.");

            using (AseTransaction trans = conn.BeginTransaction())
            {
                try
                {
                    string setClause = string.Join(", ",
                        updateParameters.Keys.Select(k => $"{k} = @{k}"));

                    string whereClause = string.Join(" AND ",
                        whereParameters.Keys.Select(k => $"{k} = @w_{k}"));

                    string sql =
                        $"UPDATE tmp_etiq_timbradas " +
                        $"SET fecha = GETDATE(), {setClause} " +
                        $"WHERE {whereClause}";

                    using (AseCommand cmd = new AseCommand(sql, conn, trans))
                    {
                        foreach (var param in updateParameters)
                            cmd.Parameters.AddWithValue("@" + param.Key, param.Value ?? DBNull.Value);

                        foreach (var param in whereParameters)
                            cmd.Parameters.AddWithValue("@w_" + param.Key, param.Value ?? DBNull.Value);

                        rowsAffected = cmd.ExecuteNonQuery();
                    }

                    if (rowsAffected > 0)
                        trans.Commit();
                    else
                        trans.Rollback();
                }
                catch (Exception ex)
                {
                    trans.Rollback();
                    LogError("Error en UpdateTimbrado", ex);
                    throw;
                }
            }
        }
    }
    catch (Exception ex)
    {
        ss_error = ex.Message;
        LogError("Error general en UpdateTimbrado", ex);
        rowsAffected = -1;
    }
    finally
    {
        _sybase.Disconnect();
    }

    return rowsAffected;
}

private void DataGridView1_CellEndEdit(object sender, DataGridViewCellEventArgs e)
{
    try
    {
        if (e.RowIndex < 0 || e.ColumnIndex < 0)
            return;

        var column = DataGridView1.Columns[e.ColumnIndex];
        if (column == null || column.Name != "hoja_marcacion")
            return;

        var row = DataGridView1.Rows[e.RowIndex];

        string nuevoHm = Convert.ToString(row.Cells["hoja_marcacion"].Value)?.Trim() ?? "";
        string idRfid = Convert.ToString(row.Cells["id_rfid"].Value)?.Trim() ?? "";
        string op = Convert.ToString(row.Cells["op"].Value)?.Trim() ?? "";
        string codTalla = Convert.ToString(row.Cells["cod_talla"].Value)?.Trim() ?? "";

        if (string.IsNullOrWhiteSpace(idRfid))
        {
            SetEstado("No se pudo actualizar: falta id_rfid.", Color.Firebrick);
            return;
        }

        var where = new Dictionary<string, object>
        {
            { "rfid", idRfid }
        };

        if (!string.IsNullOrWhiteSpace(mCodTrabajador))
            where["fotocheck"] = mCodTrabajador;

        // opcional: endurecer el WHERE
        if (!string.IsNullOrWhiteSpace(op))
            where["op"] = op;

        if (!string.IsNullOrWhiteSpace(codTalla))
            where["cod_talla"] = codTalla;

        var update = new Dictionary<string, object>
        {
            { "nhoja", nuevoHm }
        };

        int rows = _bdPrenda.UpdateTimbrado(where, update);

        if (rows > 0)
        {
            SetEstado("Hoja de marcación actualizada.", Color.SeaGreen);
            ResaltarFilaInsertada(e.RowIndex);
        }
        else
        {
            SetEstado("No se actualizó la hoja de marcación.", Color.DarkOrange);
        }
    }
    catch (Exception ex)
    {
        SetEstado("Error al actualizar HM: " + ex.Message, Color.Firebrick);
    }
}
